const records = [],
  registry = [],
  references = [],

let serial = 0;
const neutralId = () => `E${String((++serial * 47) % 997).padStart(3, "0")}`;
function context(sid, dimension) {
  const s = sources.find((x) => x.sourceId === sid);
  return {
    dimensions: [dimension],
    evidence: {
      sources: [{ id: "source", text: s.text }],
      goals: [...s.goals],
      prerequisites: [...s.prerequisites],
      desiredDepth: s.intendedDepth,
      authorityBoundaries: [
        "仅根据所附资料和明示题设评估教学与作答；不推定真实人物、现实推广效果或未展示的执行。",
      ],
      artifacts: [],
      exposure: [],
    },
  };
}
function add(record, meta, reference) {
  record.id = neutralId();
  records.push(record);
  registry.push({
    caseId: record.id,
    sourceId: meta.sourceId,
    dimension: record.dimensions[0],
    ...meta,
    recordHash: sha(record),
  });
  references.push({ caseId: record.id, ...reference });
  return record.id;
}
function keyArtifact(text) {
  return {
    id: "reference",
    text: "本题作者参考解（未向学习者提前展示）：" + text,
  };
}
function task(r, p) {
  r.evidence.question = { prompt: p.question, options: [] };
  r.evidence.artifacts.push(keyArtifact(p.key));
}

  );
